############ Reach it from inside the cluster (port 8080 -> targetPort 80) ############

$ kubectl exec curl-client -- curl -s http://web-service-clusterip:8080 | grep -o '<title>.*</title>'
<title>Welcome to nginx!</title>

$ kubectl exec curl-client -- nslookup web-service-clusterip 2>&1 | grep -A1 '^Name'
Name:	web-service-clusterip.default.svc.cluster.local
Address: 10.96.42.205

$ kubectl exec curl-client -- cat /etc/resolv.conf
search default.svc.cluster.local svc.cluster.local cluster.local
nameserver 10.96.0.10
options ndots:5

$ kubectl exec curl-client -- curl -s http://web-service-clusterip.default.svc.cluster.local:8080 -o /dev/null -w 'FQDN -> HTTP %{http_code}\n'
FQDN -> HTTP 200
