
$ kubectl apply -f 04-externalname/service.yaml
service/external-database-service created

$ kubectl get svc external-database-service
NAME                        TYPE           CLUSTER-IP   EXTERNAL-IP        PORT(S)   AGE
external-database-service   ExternalName   <none>       nencyravaliya.me   <none>    0s

$ kubectl apply -f 04-externalname/client-pod.yaml
pod/dns-test-client created

$ kubectl wait --for=condition=Ready pod/dns-test-client --timeout=120s
pod/dns-test-client condition met

############ CoreDNS answers with a CNAME - no ClusterIP, no endpoints, no kube-proxy rules ############

$ kubectl exec dns-test-client -- nslookup external-database-service 2>/dev/null | grep 'canonical name' | head -1
external-database-service.default.svc.cluster.local	canonical name = nencyravaliya.me

$ kubectl get endpointslices -l kubernetes.io/service-name=external-database-service
No resources found in default namespace.

$ docker exec kushal-lab-worker iptables -t nat -S | grep -c external-database-service
0

############ The alias is only as good as its target: nencyravaliya.me has no DNS record at the moment ############

$ kubectl exec dns-test-client -- nslookup nencyravaliya.me 2>&1 | grep -m1 -iE 'NXDOMAIN|Address: [0-9]'
** server can't find nencyravaliya.me: NXDOMAIN

$ dig +short nencyravaliya.me A | head -1; echo '(empty = no A record, checked from the Mac too)'
(empty = no A record, checked from the Mac too)

$ kubectl exec dns-test-client -- curl -s -m 5 -o /dev/null -w 'HTTP %{http_code}\n' http://external-database-service || echo 'curl exit 6 = could not resolve host (the CNAME target does not exist)'
HTTP 000
command terminated with exit code 6
curl exit 6 = could not resolve host (the CNAME target does not exist)

############ Same thing pointed at a real host, so the traffic path can be seen end to end ############

$ kubectl create service externalname external-api --external-name=api.github.com
service/external-api created

$ kubectl exec dns-test-client -- nslookup external-api 2>/dev/null | grep -E 'canonical name|^Address: [0-9]' | head -2
external-api.default.svc.cluster.local	canonical name = api.github.com
external-api.default.svc.cluster.local	canonical name = api.github.com

$ kubectl exec dns-test-client -- curl -s -m 10 -o /dev/null -w 'http://external-api  (Host: api.github.com) -> HTTP %{http_code}, connected to %{remote_ip}\n' -H 'Host: api.github.com' http://external-api
http://external-api  (Host: api.github.com) -> HTTP 301, connected to 20.207.73.85

$ kubectl exec dns-test-client -- curl -s -m 10 -o /dev/null -w 'https://external-api -> HTTP %{http_code}\n' https://external-api 2>&1 || echo '(exit 60: the TLS certificate is for api.github.com, not for the alias name - the client must use the real hostname for HTTPS)'
https://external-api -> HTTP 000
command terminated with exit code 60
(exit 60: the TLS certificate is for api.github.com, not for the alias name - the client must use the real hostname for HTTPS)

$ kubectl exec dns-test-client -- curl -s -m 10 -o /dev/null -w 'https://api.github.com directly -> HTTP %{http_code}\n' https://api.github.com
https://api.github.com directly -> HTTP 200
