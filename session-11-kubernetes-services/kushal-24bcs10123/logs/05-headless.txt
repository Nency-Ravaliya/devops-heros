############ Service name -> ALL pod IPs (no VIP in between) ############

############ Each pod has its own stable DNS name ############

$ kubectl exec headless-dns-client -- nslookup web-stateful-0.web-service-headless.default.svc.cluster.local 2>/dev/null | grep -E '^(Name|Address: 10)'
