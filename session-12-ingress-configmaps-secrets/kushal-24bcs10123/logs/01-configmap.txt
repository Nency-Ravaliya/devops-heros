############ Two ways a pod can consume it: env vars and a mounted volume ############

$ kubectl run cm-demo --image=busybox:1.36 --restart=Never --overrides='{"spec":{"containers":[{"name":"cm-demo","image":"busybox:1.36","command":["sh","-c","echo ENV: ENVIRONMENT=$ENVIRONMENT LOG_LEVEL=$LOG_LEVEL; echo FILES:; ls /etc/config; echo; cat /etc/config/DEFAULT_CURRENCY; echo"],"envFrom":[{"configMapRef":{"name":"yatri-app-config"}}],"volumeMounts":[{"name":"cfg","mountPath":"/etc/config"}]}],"volumes":[{"name":"cfg","configMap":{"name":"yatri-app-config"}}]}}'
