############ TASK 1: SOFT LINK vs HARD LINK ############
$ echo 'hello from the original file' > original.txt

$ ln -s original.txt soft.txt

$ ln original.txt hard.txt

$ ls -li
total 8
278182 -rw-r--r-- 2 root root 29 Sep  3 15:39 hard.txt
278182 -rw-r--r-- 2 root root 29 Sep  3 15:39 original.txt
278183 lrwxrwxrwx 1 root root 12 Sep  3 15:39 soft.txt -> original.txt

$ stat -c '%n  inode=%i  links=%h  type=%F' original.txt hard.txt soft.txt
original.txt  inode=278182  links=2  type=regular file
hard.txt  inode=278182  links=2  type=regular file
soft.txt  inode=278183  links=1  type=symbolic link

$ readlink soft.txt
original.txt

$ file soft.txt hard.txt
soft.txt: symbolic link to original.txt
hard.txt: ASCII text

--- editing through the hard link updates the original (same inode)
$ echo 'appended via hard link' >> hard.txt

$ cat original.txt
hello from the original file
appended via hard link

--- renaming the original: soft link breaks, hard link still works
$ mv original.txt renamed.txt

$ ls -l
total 8
-rw-r--r-- 2 root root 52 Sep  3 15:39 hard.txt
-rw-r--r-- 2 root root 52 Sep  3 15:39 renamed.txt
lrwxrwxrwx 1 root root 12 Sep  3 15:39 soft.txt -> original.txt

$ cat soft.txt
cat: soft.txt: No such file or directory

$ cat hard.txt
hello from the original file
appended via hard link

--- deleting the last 'original' name: data survives while a hard link exists
$ rm renamed.txt

$ ls -li
total 4
278182 -rw-r--r-- 1 root root 52 Sep  3 15:39 hard.txt
278183 lrwxrwxrwx 1 root root 12 Sep  3 15:39 soft.txt -> original.txt

$ cat hard.txt
hello from the original file
appended via hard link

$ cat soft.txt
cat: soft.txt: No such file or directory

--- deleting links
$ rm soft.txt hard.txt

$ ls -la
total 8
drwxr-xr-x 2 root root 4096 Sep  3 15:39 .
drwx------ 1 root root 4096 Sep  3 15:39 ..
