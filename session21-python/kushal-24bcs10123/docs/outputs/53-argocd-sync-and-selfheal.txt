  name: studytrack
  namespace: argocd
  finalizers:

  project: default
  source:
    repoURL: https://github.com/kushaltalati/devops-heros.git

    helm:
      valueFiles:
        - values.yaml
  destination:
    server: https://kubernetes.default.svc
    namespace: studytrack
  syncPolicy:
    automated:
      prune: true      # delete what is no longer in git
      selfHeal: true   # undo manual kubectl edits
    syncOptions:
      - CreateNamespace=true
