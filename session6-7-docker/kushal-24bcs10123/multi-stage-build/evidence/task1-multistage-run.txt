############ TASK 1: build and run the multi-stage Dockerfile from the course repo ############
$ cat multi-stage-dockerfile/Dockerfile
# -------------------------
# Stage 1: Build
# -------------------------
FROM node:24-alpine AS builder
WORKDIR /app
COPY package*.json ./
RUN npm install
COPY . .

# -------------------------
# Stage 2: Production
# -------------------------
FROM node:24-alpine AS production
WORKDIR /app
COPY --from=builder /app/package*.json ./
RUN npm install --omit=dev
COPY --from=builder /app/server.js ./
EXPOSE 3000
CMD ["npm", "start"]
$ docker build -t ms-hello:latest multi-stage-dockerfile 2>&1 | grep -E '^#[0-9]+ \[(builder|production)' | sed 's/ [0-9.]*s$//' | sort -u
