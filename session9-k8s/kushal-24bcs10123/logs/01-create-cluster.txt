############ CREATE CLUSTER ############

$ cat kind-cluster.yaml
# kind cluster used for sessions 9-12: one control-plane node + two worker nodes.
# The control-plane node publishes the NodePorts used by the session 10/11 labs
# and ports 80/443 for the ingress-nginx controller used in session 12, so that
# `curl http://localhost:<port>` from the Mac reaches the cluster like `minikube ip` would.
kind: Cluster
apiVersion: kind.x-k8s.io/v1alpha4
name: kushal-lab
nodes:
  - role: control-plane
    kubeadmConfigPatches:
      - |
        kind: InitConfiguration
        nodeRegistration:
          kubeletExtraArgs:
            node-labels: "ingress-ready=true"
    extraPortMappings:
      - { containerPort: 80,    hostPort: 80,    protocol: TCP }   # ingress-nginx http
      - { containerPort: 443,   hostPort: 443,   protocol: TCP }   # ingress-nginx https
      - { containerPort: 30010, hostPort: 30010, protocol: TCP }   # s10 rolling update
      - { containerPort: 30020, hostPort: 30020, protocol: TCP }   # s10 blue-green
      - { containerPort: 30030, hostPort: 30030, protocol: TCP }   # s10 canary
      - { containerPort: 30040, hostPort: 30040, protocol: TCP }   # s10 recreate
      - { containerPort: 30080, hostPort: 30080, protocol: TCP }   # s11 nodeport
  - role: worker
  - role: worker
